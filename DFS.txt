void test_output_vis(int vis[],int len)//!!vis传入的是vis[0]的指针，要额外把len传进去 
{
	printf("test_output_vis::");
	for(int i=1;i<len;i++)
		printf("%2d",vis[i]);
	printf("\n");
}

void add_undirector_edge(int u,int v,int w)
{
	edge l;
	l.v=v;
	l.weight=w;
	G[u].push_back(l);
	l.v=u;
	l.weight=w;
	G[v].push_back(l);
}

void add_director_edge(int u,int v,int w)
{
	edge l;
	l.v=v;
	l.weight=w;
	G[u].push_back(l);
}


void DFS(int u,int sum_w,int distination,int vis[],int len,int step)
{
		vis[step]=u;
	step++;
	cout<<u<<endl;//cout<<sizeof(*vis);
	if(u==distination)
	{
		if(MAX.tot_weight<sum_w)
		{
			MAX.tot_weight=sum_w;
			for(int i=1;i<len;i++)
			MAX.path[i]=vis[i];
		}
		return;
	}
	int vis_p[len];
	//cout<<sizeof(*vis_p);//test

	for(int i=1;i<len;i++)
		vis_p[i]=vis[i];
	
	
	test_output_vis(vis_p,len);
	
	for(vector<edge>::iterator it=G[u].begin();it!=G[u].end();it++)
	{
		edge tmp=*it;
		if(vis[tmp.v])
			continue;
		DFS(tmp.v,sum_w+tmp.weight,distination,vis_p,len,step);
		//cout<<""<<max_tot_weight<<endl;
	}
}
